set(TMAP_CACHE_PAGE_BYTES 4096 CACHE STRING "Maximum outdoor physical page capacity: 1024/2048/4096")
set(TMAP_PACKAGE_CACHE_PAGES 2 CACHE STRING "Fixed outdoor cache pages: 2..4")
set(TMAP_PACKAGE_SECTION_SLOTS 32 CACHE STRING "Fixed outdoor directory slots: 4..64")
if(NOT TMAP_CACHE_PAGE_BYTES MATCHES "^(1024|2048|4096)$" OR
   TMAP_PACKAGE_CACHE_PAGES LESS 2 OR TMAP_PACKAGE_CACHE_PAGES GREATER 4 OR
   TMAP_PACKAGE_SECTION_SLOTS LESS 4 OR TMAP_PACKAGE_SECTION_SLOTS GREATER 64)
  message(FATAL_ERROR "Invalid bounded outdoor storage profile")
endif()
if(TMAP_CACHE_PAGE_BYTES EQUAL 4096 AND
   (TMAP_PACKAGE_CACHE_PAGES GREATER 3 OR
    (TMAP_PACKAGE_CACHE_PAGES EQUAL 3 AND TMAP_PACKAGE_SECTION_SLOTS GREATER 40)))
  message(FATAL_ERROR "Configured outdoor workspace would exceed the legacy baseline")
endif()
